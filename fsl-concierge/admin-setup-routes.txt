function deriveSetupRoute(answers){
  const vals=Object.values(answers||{}),confidence=vals.length?vals.map(setupCertainty).reduce((a,b)=>a+b,0)/vals.length:0,reasons=[];
  const suff=Number(answers?.assessment_sufficiency?.score??0);let route=confidence>=.82&&suff>=2?'green':confidence>=.60&&suff>=1?'amber':'red';
  if(setupNoul(answers,'sensitive_work')>=.65){reasons.push('sensitive_work');if(route==='green')route='amber'}
  if(setupNoul(answers,'consequential_approval')>=.65)reasons.push('human_approval_required');
  if(suff<1.5){reasons.push('assessment_needs_clarification');route='red'}
  return{route,confidence:Number(confidence.toFixed(4)),reasons:[...new Set(reasons)]};
}
function clarificationFrom(answers,map){
  const qs=[];const uncertain=k=>{const n=setupNoul(answers,k);return n>.35&&n<.65};
  if(uncertain('multi_project'))qs.push('Do you manage more than one active project, grant, location, or programme at the same time?');
  if(uncertain('delegation'))qs.push('Do you regularly assign work to other staff and need to track it until you approve completion?');
  if(uncertain('finance_budget'))qs.push('Do you need the Concierge to monitor budgets, expenditure, forecasts, procurement, or payment-related work?');
  if(uncertain('team_performance'))qs.push('Do you need a team-performance view showing staff deliverables, progress, blockers, and follow-up?');
  if(Number(answers?.assessment_sufficiency?.score??0)<1.5)qs.push('Which three workflows should the Concierge automate or assist with first?');
  if(setupNoul(answers,'sensitive_work')>=.65&&!setupClip(map?.controls?.restrictions))qs.push('What organisational data-protection or AI-use restrictions must this Concierge follow?');
  return qs.slice(0,5);
}
