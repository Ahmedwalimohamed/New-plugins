async function sourceSendAccess(account,cred){
  if(account.provider==='gmail')return sourceRefreshAccess(account,cred);
  if(account.provider==='outlook'){
    const refresh=sourceDecrypt(cred.refresh_token_enc),body=new URLSearchParams({client_id:MICROSOFT_CLIENT_ID,client_secret:MICROSOFT_CLIENT_SECRET,refresh_token:refresh,grant_type:'refresh_token',scope:'openid profile email offline_access https://graph.microsoft.com/Mail.Read https://graph.microsoft.com/Mail.Send'});
    const r=await fetch('https://login.microsoftonline.com/common/oauth2/v2.0/token',{method:'POST',headers:{'content-type':'application/x-www-form-urlencoded'},body}),d=await r.json().catch(()=>({}));
    if(!r.ok)throw new Error(d.error_description||d.error||`Microsoft token ${r.status}`);
    if(d.refresh_token&&d.refresh_token!==refresh)await sb(`concierge_email_credentials?account_id=eq.${account.id}`,{method:'PATCH',body:JSON.stringify({refresh_token_enc:sourceEncrypt(d.refresh_token),scopes:d.scope||cred.scopes||null,token_expires_at:new Date(Date.now()+Number(d.expires_in||3600)*1000).toISOString(),updated_at:now()})});
    return d.access_token;
  }
  throw new Error('Unsupported email provider.');
}
function sourceMimeHeader(v){return String(v||'').replace(/[\r\n]+/g,' ').trim()}
async function sourceDeliverApprovedEmail(account,token,{to,subject,body,threadId}){
  if(account.provider==='gmail'){
    const mime=[`To: ${sourceMimeHeader(to)}`,`Subject: ${sourceMimeHeader(subject||'(No subject)')}`,'MIME-Version: 1.0','Content-Type: text/plain; charset="UTF-8"','Content-Transfer-Encoding: 8bit','',String(body||'')].join('\r\n');
    const payload={raw:Buffer.from(mime,'utf8').toString('base64url')};if(threadId)payload.threadId=threadId;
    const r=await fetch('https://gmail.googleapis.com/gmail/v1/users/me/messages/send',{method:'POST',headers:{authorization:`Bearer ${token}`,'content-type':'application/json'},body:JSON.stringify(payload)}),d=await r.json().catch(()=>({}));
    if(!r.ok)throw Object.assign(new Error(d.error?.message||`Gmail send ${r.status}`),{statusCode:r.status});
    return{provider:'gmail',provider_message_id:d.id||null,provider_thread_id:d.threadId||threadId||null};
  }
  if(account.provider==='outlook'){
    const payload={message:{subject:subject||'(No subject)',body:{contentType:'Text',content:String(body||'')},toRecipients:[{emailAddress:{address:to}}]},saveToSentItems:true};
    const r=await fetch('https://graph.microsoft.com/v1.0/me/sendMail',{method:'POST',headers:{authorization:`Bearer ${token}`,'content-type':'application/json'},body:JSON.stringify(payload)});
    if(!r.ok){const d=await r.json().catch(()=>({}));throw Object.assign(new Error(d.error?.message||`Outlook send ${r.status}`),{statusCode:r.status})}
    return{provider:'outlook',provider_message_id:null,provider_thread_id:threadId||null};
  }
  throw new Error('Unsupported email provider.');
}
