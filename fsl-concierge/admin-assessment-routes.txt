// --- Admin Workflow Assessment Center routes ---
const WORKFLOW_ASSESSMENT_URL='https://humanitarian-workflow-assessment.vercel.app/';
const WORKFLOW_RESPONSES_URL='https://humanitarian-workflow-responses.vercel.app/';

// Public receiver used only by the trusted assessment submission service.
// The shared secret is stored as a Railway environment variable and never sent to browsers.
if(req.method==='POST'&&url.pathname==='/api/assessment-submitted'){
  const expected=process.env.ASSESSMENT_WEBHOOK_SECRET||'';
  const supplied=String(req.headers['x-assessment-webhook-secret']||'');
  if(!expected||!supplied||!secureEqual(supplied,expected))return json(res,401,{ok:false,error:'Unauthorized assessment callback.'});
  let b={};try{const raw=await readText(req,250000);b=raw?JSON.parse(raw):{}}catch{return json(res,400,{ok:false,error:'Invalid assessment payload.'})}
  const r=b?.record||{},id=clean(r.id,36),name=clean(r.name,200),job=clean(r.job_title,200),org=clean(r.organisation,300),source=clean(r.source,80)||'legacy_humanitarian';
  if(!/^[0-9a-f-]{36}$/i.test(id)||!name||!job||!r.answers||typeof r.answers!=='object'||Array.isArray(r.answers))return json(res,400,{ok:false,error:'Incomplete assessment payload.'});
  const answers=r.answers,contactEmail=clean(r.contact_email||answers.contact_email,320).toLowerCase(),eliminate=clean(answers.eliminate_task,5000),auditWorkflows=clean(answers.audit_workflows,5000),createdAt=String(r.created_at||now());
  if(contactEmail&&!validEmail(contactEmail))return json(res,400,{ok:false,error:'Invalid contact email in assessment payload.'});
  await sb('workflow_assessment_responses?on_conflict=id',{method:'POST',headers:{Prefer:'resolution=merge-duplicates,return=minimal'},body:JSON.stringify({id,created_at:createdAt,name,job_title:job,organisation:org||null,contact_email:contactEmail||null,eliminate_task:eliminate,audit_workflows:auditWorkflows,answers,source})});

  // Submission, not an admin click, is the completion event.
  // New assessments link by contact email first; legacy submissions fall back to respondent name.
  const outstanding=await sb('concierge_assessment_invites?select=id,recipient_name,recipient_email,worker_user_id,sent_at,status&status=eq.invited&order=sent_at.desc&limit=200');
  const norm=s=>String(s||'').trim().toLocaleLowerCase().replace(/\s+/g,' ');
  const match=(outstanding||[]).find(x=>contactEmail&&norm(x.recipient_email)===norm(contactEmail))||(outstanding||[]).find(x=>norm(x.recipient_name)===norm(name));
  if(match)await sb(`concierge_assessment_invites?id=eq.${match.id}`,{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({status:'completed',completed_at:createdAt,updated_at:now()})});
  return json(res,201,{ok:true,submission_id:id,linked_invite_id:match?.id||null});
}

if(req.method==='GET'&&url.pathname==='/api/admin/assessments'){
  if(!requireAdmin(req,res,url))return;
  const [invites,responses]=await Promise.all([
    sb('concierge_assessment_invites?select=id,worker_user_id,recipient_name,recipient_email,assessment_url,status,sent_at,completed_at,notes,created_at,updated_at&order=sent_at.desc&limit=200'),
    sb('workflow_assessment_responses?select=id,created_at,name,job_title,organisation,contact_email,eliminate_task,audit_workflows,source&order=created_at.desc&limit=100')
  ]);
  return json(res,200,{ok:true,assessment_url:WORKFLOW_ASSESSMENT_URL,responses_url:WORKFLOW_RESPONSES_URL,invites:invites||[],responses:responses||[]});
}
if(req.method==='POST'&&url.pathname==='/api/admin/assessments/invite'){
  const admin=requireAdmin(req,res,url);if(!admin)return;
  const b=await readBody(req),workerId=clean(b.worker_user_id,36)||null;
  let recipientName=clean(b.recipient_name,180),recipientEmail=clean(b.recipient_email,320).toLowerCase();
  if(workerId){
    const worker=(await sb(`humanitarian_users?select=id,full_name,contact_email&id=eq.${workerId}&role=eq.worker&limit=1`))?.[0];
    if(!worker)return json(res,404,{ok:false,error:'Worker not found.'});
    recipientName=recipientName||worker.full_name||'';recipientEmail=recipientEmail||String(worker.contact_email||'').toLowerCase();
  }
  if(!recipientName&&!recipientEmail)return json(res,400,{ok:false,error:'Choose a worker or enter a recipient.'});
  if(recipientEmail&&!validEmail(recipientEmail))return json(res,400,{ok:false,error:'Enter a valid recipient email.'});
  const saved=await sb('concierge_assessment_invites',{method:'POST',headers:{Prefer:'return=representation'},body:JSON.stringify({worker_user_id:workerId,recipient_name:recipientName||null,recipient_email:recipientEmail||null,assessment_url:WORKFLOW_ASSESSMENT_URL,status:'invited',sent_by_admin:admin.username||admin.name||'admin',notes:clean(b.notes,1000)||null})});
  return json(res,201,{ok:true,invite:(saved||[])[0]||null,assessment_url:WORKFLOW_ASSESSMENT_URL});
}
const assessmentStatus=url.pathname.match(/^\/api\/admin\/assessments\/([0-9a-f-]{36})\/status$/i);
if(req.method==='POST'&&assessmentStatus){
  if(!requireAdmin(req,res,url))return;
  const b=await readBody(req),status=clean(b.status,30);if(!['invited','cancelled'].includes(status))return json(res,400,{ok:false,error:'Assessment completion is recorded automatically when the respondent submits the form.'});
  const patch={status,updated_at:now(),completed_at:null};
  const updated=await sb(`concierge_assessment_invites?id=eq.${assessmentStatus[1]}`,{method:'PATCH',headers:{Prefer:'return=representation'},body:JSON.stringify(patch)});
  return json(res,200,{ok:true,invite:(updated||[])[0]||null});
}
// --- End Admin Workflow Assessment Center routes ---
