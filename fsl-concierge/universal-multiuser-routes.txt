async function conciergeWorkerExists(workerId){
  if(!/^[0-9a-f-]{36}$/i.test(String(workerId||'')))return null;
  const rows=await sb(`humanitarian_users?select=id,username,full_name,contact_email,organization,job_title,status&id=eq.${workerId}&role=eq.worker&limit=1`);
  return (rows||[])[0]||null;
}

  return worker;
}
