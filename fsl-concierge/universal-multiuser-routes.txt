async function conciergeWorkerConfig(workerId){
  const rows=await sb(`concierge_worker_configs?select=worker_user_id,setup_profile_id,setup_version,config,status,applied_at,updated_at&worker_user_id=eq.${workerId}&status=eq.active&limit=1`);
  return (rows||[])[0]||null;
}

function conciergeModuleEnabled(config,id){const mods=config?.modules||[];if(!mods.length)return ['home','operations_memory','history'].includes(id);return mods.some(x=>x&&x.id===id&&x.enabled)}
async function conciergeWorkerExists(workerId){
  if(!/^[0-9a-f-]{36}$/i.test(String(workerId||'')))return null;
  const rows=await sb(`humanitarian_users?select=id,username,full_name,contact_email,organization,job_title,status&id=eq.${workerId}&role=eq.worker&limit=1`);
  return (rows||[])[0]||null;
}
async function conciergeLinkSetupProfile(profile,workerId){
  const worker=await conciergeWorkerExists(workerId);if(!worker)throw Object.assign(new Error('Choose a valid Concierge worker account.'),{statusCode:400});
  const t=now();
  await sb(`concierge_setup_profiles?id=eq.${profile.id}`,{method:'PATCH',body:JSON.stringify({worker_user_id:worker.id,updated_at:t})});
  if(profile.workflow_map_id)await sb(`concierge_workflow_maps?id=eq.${profile.workflow_map_id}`,{method:'PATCH',body:JSON.stringify({worker_user_id:worker.id,updated_at:t})});
  return worker;
}
function conciergeUsernameBase(name){
  const raw=String(name||'worker').normalize('NFKD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/[^a-z0-9]+/g,' ').trim();
  const parts=raw.split(/\s+/).filter(Boolean);
  return (parts.length>1?`${parts[0]}.${parts[parts.length-1]}`:parts[0]||'worker').slice(0,48);
}
async function conciergeUniqueUsername(name){
  const base=conciergeUsernameBase(name);let candidate=base;
  for(let i=0;i<50;i++){const rows=await sb(`humanitarian_users?select=id&username=eq.${encodeURIComponent(candidate)}&limit=1`);if(!(rows||[]).length)return candidate;candidate=`${base.slice(0,44)}.${i+2}`}
  return `${base.slice(0,36)}.${randomBytes(4).toString('hex')}`;
}
async function conciergeAssessmentForProfile(profile){
  return (await sb(`workflow_assessment_responses?select=id,name,job_title,organisation,contact_email,answers,created_at&id=eq.${profile.assessment_response_id}&limit=1`))?.[0]||null;
}
async function conciergeResolveContactEmail(profile,requestedEmail=''){
  const supplied=clean(requestedEmail,320).toLowerCase();
  if(supplied){if(!validEmail(supplied))throw Object.assign(new Error('Enter a valid personal/contact email.'),{statusCode:400});return supplied}
  if(profile.worker_user_id){const w=await conciergeWorkerExists(profile.worker_user_id);if(w?.contact_email)return String(w.contact_email).trim().toLowerCase()}
  const r=await conciergeAssessmentForProfile(profile),fromResponse=String(r?.contact_email||r?.answers?.contact_email||'').trim().toLowerCase();
  if(fromResponse&&validEmail(fromResponse))return fromResponse;
  return '';
}
async function conciergeCreateCandidate(profile,contactEmail){
  const response=await conciergeAssessmentForProfile(profile);if(!response)throw Object.assign(new Error('Assessment response not found.'),{statusCode:404});
  const byEmail=(await sb(`humanitarian_users?select=id,username,full_name,contact_email,organization,job_title,status&contact_email=eq.${encodeURIComponent(contactEmail)}&role=eq.worker&limit=1`))?.[0];
  if(byEmail)return byEmail;
  const username=await conciergeUniqueUsername(response.name),placeholder=hashPassword(randomBytes(32).toString('base64url'));
  const saved=await sb('humanitarian_users',{method:'POST',headers:{Prefer:'return=representation'},body:JSON.stringify({
    username,full_name:response.name,contact_email:contactEmail,organization:response.organisation||null,job_title:response.job_title||null,department:null,
    role:'worker',status:'invited',password_salt:placeholder.salt,password_hash:placeholder.hash,must_change_password:true,password_reset_status:'none'
  })});
  return (saved||[])[0];
}
function conciergeActivationUrl(token){
  const base=process.env.PUBLIC_APP_URL||`https://${process.env.RAILWAY_PUBLIC_DOMAIN||'fsl-concierge-production.up.railway.app'}`;
  return `${String(base).replace(/\/$/,'')}/activate?token=${encodeURIComponent(token)}`;
}
async function conciergeSendInviteEmail({email,name,username,url}){
  const apiKey=process.env.RESEND_API_KEY||'',from=process.env.INVITE_FROM_EMAIL||'';
  if(!apiKey||!from)return{sent:false,provider:'not_configured',error:'Transactional email is not configured.'};
  const subject='Your Operations Concierge is ready';
  const html=`<div style="font-family:Arial,sans-serif;max-width:620px;margin:auto;color:#17242d"><div style="padding:24px 0;border-bottom:1px solid #dce3e8"><div style="font-size:12px;letter-spacing:.12em;text-transform:uppercase;color:#557684;font-weight:700">Humanitarian Operations Concierge</div><h1 style="font-size:26px;margin:7px 0 0">Your Concierge is ready</h1></div><div style="padding:26px 0"><p>Hello ${esc(name)},</p><p>Your workflow assessment has been reviewed and your Operations Concierge has been prepared.</p><p><b>Username:</b> ${esc(username)}</p><p style="margin:26px 0"><a href="${esc(url)}" style="background:#0f6c8e;color:#fff;text-decoration:none;padding:13px 18px;border-radius:8px;font-weight:700">Activate your Concierge</a></p><p>This secure activation link expires in 48 hours. You will create your own password; no password is sent by email.</p><p style="color:#687984;font-size:13px">After activation, you will choose which work email and document sources to connect. Your contact email is not read as a work source unless you explicitly connect it later.</p></div></div>`;
  try{
    const r=await fetch('https://api.resend.com/emails',{method:'POST',headers:{authorization:`Bearer ${apiKey}`,'content-type':'application/json'},body:JSON.stringify({from,to:[email],subject,html})});
    const d=await r.json().catch(()=>({}));if(!r.ok)return{sent:false,provider:'resend',error:clean(d?.message||`Resend ${r.status}`,500)};
    return{sent:true,provider:'resend',id:d.id||null,error:null};
  }catch(e){return{sent:false,provider:'resend',error:clean(e.message,500)}}
}
async function conciergeIssueInvitation(profile,worker,contactEmail){
  await sb(`concierge_worker_invitations?worker_user_id=eq.${worker.id}&status=in.(pending,sent,delivery_failed)`,{method:'PATCH',body:JSON.stringify({status:'cancelled',updated_at:now()})});
  const token=randomBytes(32).toString('base64url'),tokenHash=sign(`invite:${token}`),expires=new Date(Date.now()+48*60*60*1000).toISOString(),activationUrl=conciergeActivationUrl(token);
  const saved=await sb('concierge_worker_invitations',{method:'POST',headers:{Prefer:'return=representation'},body:JSON.stringify({
    assessment_response_id:profile.assessment_response_id,setup_profile_id:profile.id,worker_user_id:worker.id,contact_email:contactEmail,username:worker.username,token_hash:tokenHash,status:'pending',expires_at:expires
  })}),inv=(saved||[])[0];
  const delivery=await conciergeSendInviteEmail({email:contactEmail,name:worker.full_name,username:worker.username,url:activationUrl}),t=now(),status=delivery.sent?'sent':delivery.provider==='not_configured'?'pending':'delivery_failed';
  await sb(`concierge_worker_invitations?id=eq.${inv.id}`,{method:'PATCH',body:JSON.stringify({status,sent_at:delivery.sent?t:null,delivery_provider:delivery.provider,delivery_id:delivery.id||null,delivery_error:delivery.error||null,updated_at:t})});
  return{invitation_id:inv.id,status,expires_at:expires,activation_url:activationUrl,delivery};
}

if(req.method==='POST'&&url.pathname==='/activate'){
  const form=new URLSearchParams(await readText(req)),token=clean(form.get('token'),200),password=String(form.get('password')||''),confirm=String(form.get('confirm')||'');
  const tokenHash=sign(`invite:${token}`),inv=(await sb(`concierge_worker_invitations?select=*&token_hash=eq.${encodeURIComponent(tokenHash)}&limit=1`))?.[0];
  if(!inv||!['pending','sent','delivery_failed'].includes(inv.status)||new Date(inv.expires_at).getTime()<Date.now())return htmlResponse(res,authShell('Activation link expired',`<div class="card"><h2>Activation link expired</h2><p class="note">Ask the Concierge administrator to send a new invitation.</p></div>`));

  if(password.length<8||password!==confirm)return htmlResponse(res,activationPage(token,worker,'Passwords must match and contain at least 8 characters.'));
  const hp=hashPassword(password),t=now();
  await sb(`humanitarian_users?id=eq.${worker.id}`,{method:'PATCH',body:JSON.stringify({password_salt:hp.salt,password_hash:hp.hash,must_change_password:false,status:'active',password_reset_status:'none',updated_at:t})});

  return redirect(res,'/',sessionCookie(makeSession({role:'worker',user_id:worker.id,name:worker.full_name,username:worker.username,must_change:false})));
